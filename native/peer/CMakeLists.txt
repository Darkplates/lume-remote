cmake_minimum_required(VERSION 3.21)
project(LumePeerDependencies LANGUAGES C CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT LUME_PEER_SOURCE_ROOT)
  message(FATAL_ERROR "Set LUME_PEER_SOURCE_ROOT to the pinned libdatachannel and mbedtls source directory")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(MBEDTLS_SSL_DTLS_SRTP)
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(USE_SHARED_MBEDTLS_LIBRARY OFF CACHE BOOL "" FORCE)
set(USE_STATIC_MBEDTLS_LIBRARY ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory("${LUME_PEER_SOURCE_ROOT}/mbedtls" mbedtls)
add_library(lume_mbedtls INTERFACE)
target_link_libraries(lume_mbedtls INTERFACE mbedtls mbedx509 mbedcrypto)
add_library(MbedTLS::MbedTLS ALIAS lume_mbedtls)
set(USE_MBEDTLS ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_DEPS_LIBS OFF CACHE BOOL "" FORCE)
set(NO_MEDIA ON CACHE BOOL "" FORCE)
set(NO_WEBSOCKET ON CACHE BOOL "" FORCE)
set(NO_EXAMPLES ON CACHE BOOL "" FORCE)
set(NO_TESTS ON CACHE BOOL "" FORCE)
add_subdirectory("${LUME_PEER_SOURCE_ROOT}/libdatachannel" libdatachannel)
